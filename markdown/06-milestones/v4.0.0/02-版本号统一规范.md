  | grep -v "node_modules" | grep -v "target" | grep -v "markdown/06-milestones/v3.7"
# 应返回 0

grep -r "3\.7\.2" --include="*.toml" --include="*.json" --include="*.html" . \
  | grep -v "node_modules" | grep -v "target" | grep -v "markdown/06-milestones/v3.7"
