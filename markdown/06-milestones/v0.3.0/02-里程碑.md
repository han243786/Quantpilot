### M1-6 综合验证

**验收**：以下策略可编译 + 运行 + 回测：
```qs
fn strategy() {
    let closes = fetch("BTCUSDT", interval="1d", lookback=200)?
    let volumes = fetch_volume("BTCUSDT", interval="1d", lookback=200)?
    let highs = fetch_high("BTCUSDT", interval="1d", lookback=200)?
    let lows = fetch_low("BTCUSDT", interval="1d", lookback=200)?
    let fast = ema(closes, 10)
    let slow = ema(closes, 30)
    let atr14 = atr(highs, lows, closes, 14)
    let bb = bollinger(closes, 20, 2.0)
    let obv_rising = obv(closes, volumes) > obv(closes, volumes)[1]
    if fast > slow && closes.last() > bb.lower && obv_rising {
        emit Intent("BUY", instrument="BTCUSDT", quantity=0.01 * (1.0 / (atr14 / closes.last())))
    }
}
```
